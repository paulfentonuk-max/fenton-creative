Quick Start

// Install ShieldPost SDK
npm install @fentoncreative/shieldpost

// Initialize client
import { ShieldPost } from '@fentoncreative/shieldpost';

const shield = new ShieldPost({
  apiKey: process.env.SHIELDPOST_KEY,
  region: 'eu-west'
});
